5- Le thème

Dans Theme.kt :

private val DarkColorScheme = darkColorScheme(
    background = Blue800,
    surface = Blue600,
    primary = Blue300,
    primaryContainer = Blue600,
    secondary = White,
    onSecondary = White,
    tertiary = Gray500
)

private val LightColorScheme = lightColorScheme(
    background = Gray400,
    surface = White,
    primary = Blue400,
    primaryContainer = Blue100,
    secondary = Black,
    onSecondary = White,
    tertiary = Gray700
)

@Composable
fun MyWeatherTheme(
    darkTheme:Bool = isSystemInDarkTheme
    content:Composable () -> Unit
) {
    val colorScheme = if (darkTheme) {
        DarkColorScheme
    } else {
        LightColorScheme
    }

    MaterialTheme(
        colorScheme = colorScheme
        typography = Typography
        content = content
    )
}
